function randomEmail() {
    const idx = Math.floor(Math.random() * emailData.length);
    currentIndex = idx;
    renderEmail(idx);
}

function renderEmail(index) {
    const data = emailData[index];
    document.getElementById('badge').textContent = '#' + data.id;
    document.getElementById('currentNum').textContent = data.id;
    document.getElementById('subjectDisplay').textContent = data.subject;
    document.getElementById('bodyDisplay').textContent = data.body;
}

function copyText() {
    const body = document.getElementById('bodyDisplay').textContent;
    navigator.clipboard.writeText(body).then(function() {
        copyCount++;
        document.getElementById('copyCount').textContent = copyCount;
        alert('✅ متن کپی شد!');
    }).catch(function() {
        alert('❌ خطا در کپی. لطفاً دستی کپی کنید.');
    });
}

function sendEmail() {
    const subject = encodeURIComponent(document.getElementById('subjectDisplay').textContent);
    const body = encodeURIComponent(document.getElementById('bodyDisplay').textContent);
    window.location.href = 'mailto:abuse@telegram.org?subject=' + subject + '&body=' + body;
    sendCount++;
    document.getElementById('sendCount').textContent = sendCount;
}

function sendGmail() {
    const subject = encodeURIComponent(document.getElementById('subjectDisplay').textContent);

    sendCount++;
    document.getElementById('sendCount').textContent = sendCount;
}
